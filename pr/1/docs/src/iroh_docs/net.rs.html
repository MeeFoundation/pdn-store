25/// Bound on one whole sync exchange, connection establishment included.
26///
27/// Nothing below this point carries a timeout of its own, so a peer that
28/// stays connected and stops talking stalls forever. Two things ride on the
29/// bound. The live actor tracks one running exchange per namespace and peer
30/// and refuses to start another while one runs, so a stalled exchange
31/// blocks that pair and drops every later sync trigger silently. And a
32/// session holds a store snapshot, whose read transaction holds back
33/// reclamation of every page freed while it lives — of the oldest live one,
34/// so concurrent sessions cost the same window as a single one, and the
35/// window is this bound.
36///
37/// The bound is on the exchange as a whole, not on the wait between
38/// messages, and there is no shorter liveness bound beside it. A peer
39/// sending one message every few seconds defeats a between-messages bound
40/// while holding both resources. A connection that goes dead rather than
41/// quiet — a phone entering a tunnel — is already cut below, by QUIC keep
42/// alives against its idle timeout, well inside this bound. And a
43/// between-messages bound could not tell a slow transfer from silence
44/// anyway: a message is delivered whole, so a peer sending one large
45/// message for minutes looks exactly like a peer sending nothing.
46///
47/// The value covers a first sync of a large store over a slow link: an
48/// entry is roughly 280 bytes on the wire, and a peer holding nothing
49/// receives the served set in one message, so 10,000 entries are about 2.8
50/// MB — five minutes carry that from about 75 kbit/s up. Beyond that the
51/// exchange cannot complete at all rather than completing slowly, because a
52/// message is ingested whole or not at all: a cut mid-message delivers
53/// nothing, and the next session starts over. Raising the bound moves that
54/// cliff; only bounding the size of a transmitted set removes it.
55pub const SYNC_SESSION_TIMEOUT: Duration = Duration::from_secs(300);

57/// Connect to a peer and sync a replica.
58///
59/// `filter` is this side's egress filter for the session: what the dialing
60/// node reveals of its replica to the peer. `None` serves the full view.
61///
62/// The session serves entries from a store snapshot frozen at session
63/// setup, so the served view is stable across the exchange; entries
64/// written meanwhile travel on the next session. The snapshot is released
65/// when the session ends, on every path. The whole exchange is bounded by
66/// [`SYNC_SESSION_TIMEOUT`].
67pub async fn connect_and_sync(
68    endpoint: &Endpoint,
69    sync: &SyncHandle,
70    namespace: NamespaceId,
71    peer: EndpointAddr,
72    metrics: Option<&Metrics>,
73    filter: Option<crate::filter::EntryFilter>,
74) -> Result<SyncFinished, ConnectError> {
75    match time::timeout(
76        SYNC_SESSION_TIMEOUT,
77        connect_and_sync_inner(endpoint, sync, namespace, peer, metrics, filter),
78    )
79    .await
80    {
81        Ok(res) => res,
82        Err(_elapsed) => Err(ConnectError::connect(anyhow::anyhow!(
83            "sync exchange timed out after {SYNC_SESSION_TIMEOUT:?}"
84        ))),
85    }
86}

88async fn connect_and_sync_inner(
89    endpoint: &Endpoint,
90    sync: &SyncHandle,
91    namespace: NamespaceId,
92    peer: EndpointAddr,
93    metrics: Option<&Metrics>,
94    filter: Option<crate::filter::EntryFilter>,
95) -> Result<SyncFinished, ConnectError> {
96    let t_start = Instant::now();
97    let peer_id = peer.id;
98    trace!("connect");
99    let connection = endpoint
100        .connect(peer, crate::ALPN)
101        .await
102        .map_err(ConnectError::connect)?;
103
104    let (mut send_stream, mut recv_stream) =
105        connection.open_bi().await.map_err(ConnectError::connect)?;
106
107    let t_connect = t_start.elapsed();
108    debug!(?t_connect, "connected");
109
110    let res = run_alice(
111        &mut send_stream,
112        &mut recv_stream,
113        sync,
114        namespace,
115        peer_id,
116        filter,
117    )
118    .await;
119
120    send_stream.finish().map_err(ConnectError::close)?;
121    send_stream.stopped().await.map_err(ConnectError::close)?;
122    recv_stream
123        .read_to_end(0)
124        .await
125        .map_err(ConnectError::close)?;
126
127    if let Some(metrics) = metrics {
128        if res.is_ok() {
129            metrics.sync_via_connect_success.inc();
130        } else {
131            metrics.sync_via_connect_failure.inc();
132        }
133    }
134
135    let t_process = t_start.elapsed() - t_connect;
136    match &res {
137        Ok(res) => {
138            debug!(
139                ?t_connect,
140                ?t_process,
141                sent = %res.num_sent,
142                recv = %res.num_recv,
143                "done, ok"
144            );
145        }
146        Err(err) => {
147            debug!(?t_connect, ?t_process, ?err, "done, failed");
148        }
149    }
150
151    let outcome = res?;
152
153    let timings = Timings {
154        connect: t_connect,
155        process: t_process,
156    };
157
158    let res = SyncFinished {
159        namespace,
160        peer: peer_id,
161        outcome,
162        timings,
163    };
164
165    Ok(res)
166}

168/// Whether we want to accept or reject an incoming sync request.
169#[derive(Clone)]
170pub enum AcceptOutcome {
171    /// Accept the sync request.
172    Allow {
173        /// This side's egress filter for the session: what the serving
174        /// node reveals of the replica to this peer. `None` serves the
175        /// full view.
176        filter: Option<crate::filter::EntryFilter>,
177    },
178    /// Decline the sync request
179    Reject(AbortReason),
180}
181
182impl std::fmt::Debug for AcceptOutcome {
183    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
184        match self {
185            AcceptOutcome::Allow { filter: None } => write!(f, "Allow"),
186            AcceptOutcome::Allow { filter: Some(_) } => write!(f, "Allow(filtered)"),
187            AcceptOutcome::Reject(reason) => write!(f, "Reject({reason:?})"),
188        }
189    }
190}

192/// Handle an iroh-docs connection and sync all shared documents in the replica store.
193///
194/// An allowed session serves entries from a store snapshot frozen right
195/// after the accept decision (see [`connect_and_sync`] for the snapshot
196/// semantics); a rejected request never opens one. The whole exchange is
197/// bounded by [`SYNC_SESSION_TIMEOUT`], mirroring [`connect_and_sync`]: a
198/// stalled accept blocks the pair just the same.
199pub async fn handle_connection<F, Fut>(
200    sync: SyncHandle,
201    connection: iroh::endpoint::Connection,
202    accept_cb: F,
203    metrics: Option<&Metrics>,
204) -> Result<SyncFinished, AcceptError>
205where
206    F: Fn(NamespaceId, PublicKey) -> Fut,
207    Fut: Future<Output = AcceptOutcome>,
208{
209    let peer = connection.remote_id();
210    // A timeout has to name the namespace to release the pair the accept
211    // decision registered as running — an error without one is routed as a
212    // failure before the first message and leaves the pair running for
213    // good. Named when the decision is asked for rather than when it comes
214    // back, because the bound can be reached while it is still running.
215    let accepted: Arc<OnceLock<NamespaceId>> = Default::default();
216    let observed_accept_cb = {
217        let accepted = Arc::clone(&accepted);
218        move |namespace, peer| {
219            let _ = accepted.set(namespace);
220            accept_cb(namespace, peer)
221        }
222    };
223
224    match time::timeout(
225        SYNC_SESSION_TIMEOUT,
226        handle_connection_inner(sync, connection, observed_accept_cb, metrics),
227    )
228    .await
229    {
230        Ok(res) => res,
231        Err(_elapsed) => Err(AcceptError::sync(
232            peer,
233            accepted.get().copied(),
234            anyhow::anyhow!("sync exchange timed out after {SYNC_SESSION_TIMEOUT:?}"),
235        )),
236    }
237}
238
239async fn handle_connection_inner<F, Fut>(
240    sync: SyncHandle,
241    connection: iroh::endpoint::Connection,
242    accept_cb: F,
243    metrics: Option<&Metrics>,
244) -> Result<SyncFinished, AcceptError>
245where
246    F: Fn(NamespaceId, PublicKey) -> Fut,
247    Fut: Future<Output = AcceptOutcome>,
248{
249    let t_start = Instant::now();
250    let peer = connection.remote_id();
251    let (mut send_stream, mut recv_stream) = connection
252        .accept_bi()
253        .await
254        .map_err(|e| AcceptError::open(peer, e))?;
255
256    let t_connect = t_start.elapsed();
257    let span = error_span!("accept", peer = %peer.fmt_short(), namespace = tracing::field::Empty);
258    span.in_scope(|| {
259        debug!(?t_connect, "connection established");
260    });
261
262    let mut state = BobState::new(peer);
263    let res = state
264        .run(&mut send_stream, &mut recv_stream, sync, accept_cb)
265        .instrument(span.clone())
266        .await;
267
268    if let Some(metrics) = metrics {
269        if res.is_ok() {
270            metrics.sync_via_accept_success.inc();
271        } else {
272            metrics.sync_via_accept_failure.inc();
273        }
274    }
275
276    let namespace = state.namespace();
277    let outcome = state.into_outcome();
278
279    // The exchange's own result wins: teardown fails as a consequence of
280    // whatever ended the exchange, so reporting the consequence buries the
281    // cause the caller routes on. Nothing after the rounds may replace the
282    // first error — the serving side's terminal frame swallows its own for
283    // the same reason.
284    let closed: Result<(), AcceptError> = async {
285        send_stream
286            .finish()
287            .map_err(|error| AcceptError::close(peer, namespace, error))?;
288        send_stream
289            .stopped()
290            .await
291            .map_err(|error| AcceptError::close(peer, namespace, error))?;
292        recv_stream
293            .read_to_end(0)
294            .await
295            .map_err(|error| AcceptError::close(peer, namespace, error))?;
296        Ok(())
297    }
298    .await;
299
300    let t_process = t_start.elapsed() - t_connect;
301    span.in_scope(|| match &res {
302        Ok(_res) => {
303            debug!(
304                ?t_connect,
305                ?t_process,
306                sent = %outcome.num_sent,
307                recv = %outcome.num_recv,
308                "done, ok"
309            );
310        }
311        Err(err) => {
312            debug!(?t_connect, ?t_process, ?err, "done, failed");
313        }
314    });
315
316    let namespace = res.and_then(|namespace| closed.map(|()| namespace))?;
317
318    let timings = Timings {
319        connect: t_connect,
320        process: t_process,
321    };
322    let res = SyncFinished {
323        namespace,
324        outcome,
325        peer,
326        timings,
327    };
328
329    Ok(res)
330}

332/// Details of a finished sync operation.
333#[derive(Debug, Clone)]
334pub struct SyncFinished {
335    /// The namespace that was synced.
336    pub namespace: NamespaceId,
337    /// The peer we syned with.

341    /// The time this operation took
342    pub timings: Timings,

345/// Time a sync operation took

354/// Errors that may occur on handling incoming sync connections.
355#[derive(thiserror::Error, Debug)]
356#[allow(missing_docs)]
357pub enum AcceptError {
358    /// Failed to establish connection
359    #[error("Failed to establish connection")]
360    Connect {
361        #[source]
362        error: anyhow::Error,
363    },
364    /// Failed to open replica
365    #[error("Failed to open replica with {peer:?}")]
366    Open {
367        peer: PublicKey,
368        #[source]
369        error: anyhow::Error,
370    },
371    /// We aborted the sync request.
372    #[error("Aborted sync of {namespace:?} with {peer:?}: {reason:?}")]
373    Abort {
374        peer: PublicKey,
375        namespace: NamespaceId,
376        reason: AbortReason,
377    },
378    /// Failed to run sync
379    #[error("Failed to sync {namespace:?} with {peer:?}")]
380    Sync {
381        peer: PublicKey,
382        namespace: Option<NamespaceId>,
383        #[source]
384        error: anyhow::Error,
385    },
386    /// Failed to close
387    #[error("Failed to close {namespace:?} with {peer:?}")]
388    Close {
389        peer: PublicKey,
390        namespace: Option<NamespaceId>,
391        #[source]
392        error: anyhow::Error,
393    },
394}

396/// Errors that may occur on outgoing sync requests.
397#[derive(thiserror::Error, Debug)]
398#[allow(missing_docs)]
399pub enum ConnectError {
400    /// Failed to establish connection
401    #[error("Failed to establish connection")]
402    Connect {
403        #[source]
404        error: anyhow::Error,
405    },
406    /// The remote peer aborted the sync request.
407    #[error("Remote peer aborted sync: {0:?}")]
408    RemoteAbort(AbortReason),
409    /// Failed to run sync
410    #[error("Failed to sync")]
411    Sync {
412        #[source]
413        error: anyhow::Error,
414    },
415    /// Failed to close
416    #[error("Failed to close connection1")]
417    Close {
418        #[source]
419        error: anyhow::Error,
420    },
421}

423/// Reason why we aborted an incoming sync request.
424#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, Eq)]
425pub enum AbortReason {
426    /// Namespace is not available.
427    NotFound,
428    /// We are already syncing this namespace.
429    AlreadySyncing,
430    /// We experienced an error while trying to provide the requested resource
431    InternalServerError,
432}

434impl AcceptError {
435    fn open(peer: PublicKey, error: impl Into<anyhow::Error>) -> Self {
436        Self::Open {
437            peer,
438            error: error.into(),
439        }
440    }
441    pub(crate) fn sync(
442        peer: PublicKey,
443        namespace: Option<NamespaceId>,
444        error: impl Into<anyhow::Error>,
445    ) -> Self {
446        Self::Sync {
447            peer,
448            namespace,
449            error: error.into(),
450        }
451    }
452    fn close(
453        peer: PublicKey,
454        namespace: Option<NamespaceId>,
455        error: impl Into<anyhow::Error>,
456    ) -> Self {
457        Self::Close {
458            peer,
459            namespace,
460            error: error.into(),
461        }
462    }
463    /// Get the peer's node ID (if available)
464    pub fn peer(&self) -> Option<PublicKey> {
465        match self {
466            AcceptError::Connect { .. } => None,
467            AcceptError::Open { peer, .. } => Some(*peer),
468            AcceptError::Sync { peer, .. } => Some(*peer),
469            AcceptError::Close { peer, .. } => Some(*peer),
470            AcceptError::Abort { peer, .. } => Some(*peer),
471        }
472    }
473
474    /// Get the namespace (if available)
475    pub fn namespace(&self) -> Option<NamespaceId> {
476        match self {
477            AcceptError::Connect { .. } => None,
478            AcceptError::Open { .. } => None,
479            AcceptError::Sync { namespace, .. } => namespace.to_owned(),
480            AcceptError::Close { namespace, .. } => namespace.to_owned(),
481            AcceptError::Abort { namespace, .. } => Some(*namespace),
482        }
483    }
484}

493        Self::Close {
