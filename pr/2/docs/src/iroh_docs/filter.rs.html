1//! Session-scoped egress access control for reconciliation.
2//!
3//! The reconciliation engine reads entries only through the ranger store
4//! trait: the crate-internal `SessionStore` wraps a replica's store with an optional per-session
5//! [`EntryFilter`], so every read the engine performs — the first key, range
6//! iterations, fingerprints, counts — sees only admitted entries. Filtering
7//! the iterators filters the fingerprints by construction (fingerprints are
8//! computed by iterating ranges), so no unadmitted entry can leak through a
9//! fingerprint, a split boundary, or an item transmission. Write-side
10//! methods (`entry_put`, `prefixes_of`, `remove_prefix_filtered`) pass
11//! through unfiltered — ingest is the `validate_entry` hook's concern.
12//!
13//! The domain meaning of a filter (grants, identities) stays outside this
14//! crate: an embedder hands in opaque predicates over [`SignedEntry`]
15//! through a [`SessionAccessProvider`], consulted per session on both
16//! session roles — accepting an incoming sync request and dialing out —
17//! because both ends of a reconciliation serve entries.
18

30/// Per-session egress predicate: `true` admits the entry into the peer's
31/// view. Must be cheap — it runs on every entry a range scan touches.
32pub type EntryFilter = Arc<dyn Fn(&SignedEntry) -> bool + Send + Sync + 'static>;
33
34/// What one session may see of a replica, decided per (namespace, peer) at
35/// session setup and frozen for the session.
36#[derive(Clone)]
37pub enum SessionAccess {
38    /// The peer sees the replica whole.
39    Full,
40    /// The peer sees only the entries the filter admits.
41    Filtered(EntryFilter),
42    /// No session: reject as if the replica were not hosted here.
43    Deny,
44}
45
46impl std::fmt::Debug for SessionAccess {
47    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
48        match self {
49            SessionAccess::Full => write!(f, "Full"),
50            SessionAccess::Filtered(_) => write!(f, "Filtered(..)"),
51            SessionAccess::Deny => write!(f, "Deny"),
52        }
53    }
54}

56/// Which end of the session this node is on when the provider is asked.
57///
58/// Both ends serve entries (reconciliation is bidirectional), but the
59/// policies differ: a node may refuse to *accept* a caller it cannot judge
60/// while still being allowed to *dial* out with a closed egress (serving
61/// nothing, receiving whatever the callee's own filter admits).
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub enum SessionRole {
64    /// This node is accepting an incoming sync request.
65    Accept,
66    /// This node is dialing out to sync.
67    Dial,
68}
69
70/// Future returned by a [`SessionAccessProvider`].
71pub type SessionAccessFuture = Pin<Box<dyn Future<Output = SessionAccess> + Send + 'static>>;
72
73/// Decides, per session, what a peer may see of a namespace. Consulted on
74/// both session roles. `None` (provider unset) keeps every session
75/// [`SessionAccess::Full`] — vanilla iroh-docs behaviour.
76pub type SessionAccessProvider =
77    Arc<dyn Fn(NamespaceId, PublicKey, SessionRole) -> SessionAccessFuture + Send + Sync + 'static>;
78
79/// A replica store narrowed to one session's view.
80///
81/// With `filter: None` every method delegates unchanged. With a filter, all
82/// reading methods yield only admitted entries; `get_first` returns the
83/// first *admitted* key (the first physical key would leak an unadmitted
84/// entry's existence through the initial range boundary), and
85/// `get_fingerprint` recomputes over the filtered range iterator.
86pub(crate) struct SessionStore<S> {
87    inner: S,
88    filter: Option<EntryFilter>,
89}
90
91impl<S> SessionStore<S> {
92    pub(crate) fn new(inner: S, filter: Option<EntryFilter>) -> Self {
93        Self { inner, filter }
94    }
95}
96
97/// Whether `entry` passes the session's filter (no filter admits all).
98fn admitted(filter: &Option<EntryFilter>, entry: &SignedEntry) -> bool {
99    match filter {
100        None => true,
101        Some(f) => f(entry),
102    }
103}
104
105impl<S: Store<SignedEntry>> Store<SignedEntry> for SessionStore<S> {
106    type Error = S::Error;
107    type RangeIterator<'a>
108        = FilteredIter<S::RangeIterator<'a>>
109    where
110        S: 'a;
111    type ParentIterator<'a>
112        = S::ParentIterator<'a>
113    where
114        S: 'a;
115
116    fn get_first(&mut self) -> Result<RecordIdentifier, Self::Error> {
117        let Some(filter) = self.filter.clone() else {
118            return self.inner.get_first();
119        };
120        // The full range (x == y wraps around); the first admitted key, or
121        // the default when nothing is admitted — indistinguishable from an
122        // empty replica, exactly as intended.
123        let all = Range::new(RecordIdentifier::default(), RecordIdentifier::default());
124        for entry in self.inner.get_range(all)? {
125            let entry = entry?;
126            if filter(&entry) {
127                return Ok(entry.id().clone());
128            }
129        }
130        Ok(RecordIdentifier::default())
131    }
132
133    #[cfg(test)]
134    fn get(&mut self, key: &RecordIdentifier) -> Result<Option<SignedEntry>, Self::Error> {
135        let found = self.inner.get(key)?;
136        Ok(found.filter(|e| admitted(&self.filter, e)))
137    }
138
139    #[cfg(test)]
140    fn len(&mut self) -> Result<usize, Self::Error> {
141        let all = Range::new(RecordIdentifier::default(), RecordIdentifier::default());
142        self.get_range_len(all)
143    }
144
145    #[cfg(test)]
146    fn is_empty(&mut self) -> Result<bool, Self::Error> {
147        Ok(self.len()? == 0)
148    }
149
150    fn get_fingerprint(
151        &mut self,
152        range: &Range<RecordIdentifier>,
153    ) -> Result<Fingerprint, Self::Error> {
154        if self.filter.is_none() {
155            return self.inner.get_fingerprint(range);
156        }
157        // Recomputed over the session's own (filtered) range iterator, the
158        // same way the store computes it over its full one.
159        let mut fp = Fingerprint::empty();
160        for entry in self.get_range(range.clone())? {
161            fp ^= entry?.as_fingerprint();
162        }
163        Ok(fp)
164    }
165
166    fn entry_put(&mut self, entry: SignedEntry) -> Result<(), Self::Error> {
167        self.inner.entry_put(entry)
168    }
169
170    fn get_range(
171        &mut self,
172        range: Range<RecordIdentifier>,
173    ) -> Result<Self::RangeIterator<'_>, Self::Error> {
174        Ok(FilteredIter {
175            inner: self.inner.get_range(range)?,
176            filter: self.filter.clone(),
177        })
178    }
179
180    #[cfg(test)]
181    fn prefixed_by(
182        &mut self,
183        prefix: &RecordIdentifier,
184    ) -> Result<Self::RangeIterator<'_>, Self::Error> {
185        Ok(FilteredIter {
186            inner: self.inner.prefixed_by(prefix)?,
187            filter: self.filter.clone(),
188        })
189    }
190
191    fn prefixes_of(
192        &mut self,
193        key: &RecordIdentifier,
194    ) -> Result<Self::ParentIterator<'_>, Self::Error> {
195        self.inner.prefixes_of(key)
196    }
197
198    #[cfg(test)]
199    fn all(&mut self) -> Result<Self::RangeIterator<'_>, Self::Error> {
200        Ok(FilteredIter {
201            inner: self.inner.all()?,
202            filter: self.filter.clone(),
203        })
204    }
205
206    #[cfg(test)]
207    fn entry_remove(&mut self, key: &RecordIdentifier) -> Result<Option<SignedEntry>, Self::Error> {
208        self.inner.entry_remove(key)
209    }
210
211    fn remove_prefix_filtered(
212        &mut self,
213        prefix: &RecordIdentifier,
214        predicate: impl Fn(&crate::sync::Record) -> bool,
215    ) -> Result<usize, Self::Error> {
216        self.inner.remove_prefix_filtered(prefix, predicate)
217    }
218}

226/// A range iterator narrowed to admitted entries; errors pass through.
227pub(crate) struct FilteredIter<I> {
228    inner: I,
229    filter: Option<EntryFilter>,
230}
231
232impl<I, E> Iterator for FilteredIter<I>
233where
234    I: Iterator<Item = Result<SignedEntry, E>>,
235{
236    type Item = Result<SignedEntry, E>;
237
238    fn next(&mut self) -> Option<Self::Item> {
239        loop {
240            match self.inner.next()? {
241                Err(e) => return Some(Err(e)),
242                Ok(entry) => {
243                    if admitted(&self.filter, &entry) {
244                        return Some(Ok(entry));
245                    }
246                }
247            }
248        }
249    }
250}
